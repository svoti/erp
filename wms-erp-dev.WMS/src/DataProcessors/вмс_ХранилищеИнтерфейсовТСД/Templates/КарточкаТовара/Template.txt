Функция ВывестиПоляТовара(ДанныеТовара)
    
    ДанныеТовараМассив = Новый Массив;
    
	Если ДанныеТовара=Неопределено Тогда
		ДанныеТовараМассив.Добавить("&amp;lt;не указан&amp;gt;");
	Иначе
		ДопустимыеПоля = Новый Массив;
		ДопустимыеПоля.Добавить("Товар");  
		ДопустимыеПоля.Добавить("Характиристика"); 
		ДопустимыеПоля.Добавить("Серия"); 
		
		//ДопустимыеПоля.Добавить("Характеристика");
		Для Каждого ПолеТовара Из ДанныеТовара Цикл 
		    Ключ = ПолеТовара.Ключ; 
			НайденноеПоле = ДопустимыеПоля.Найти(Ключ);
			Если НайденноеПоле &amp;lt;&amp;gt; Неопределено Тогда      
				Значение = ПолеТовара.Значение;
				ДанныеТовараМассив.Добавить(Значение);
			КонецЕсли;
		КонецЦикла;
	КонецЕсли;
	
	Если ДанныеТовараМассив.Количество() &amp;gt; 1 Тогда
		ДобавитьПоляТовара(ДанныеТовараМассив.Количество());
	КонецЕсли;
	
	НомерПоля = 1;
	Для Каждого ПолеТовара Из ДанныеТовараМассив Цикл
		ИмяПоля = "ПолеТовара" + Формат(НомерПоля, "ЧГ=0");
		ЭтаФорма[ИмяПоля] = ПолеТовара;
		НомерПоля = НомерПоля + 1;
	КонецЦикла;

	Элементы.ПолеТовара1.Шрифт = Новый Шрифт(Элементы.ПолеТовара1.Шрифт, , , Истина);

	Возврат Истина;

Функция ДобавитьПоляТовара(Количество)

	НовыеРеквизиты = Новый Массив;
	
	ТипПоля = Новый ОписаниеТипов("Строка");
	ШаблонПоля = Элементы.ПолеТовара1;

	Для НомерПоля=2 По Количество Цикл
		ИмяПоля = "ПолеТовара" + Формат(НомерПоля, "ЧГ=0");  
		НайденныйЭлемент = ЭтотОбъект.Элементы.Найти(ИмяПоля);
		Если НайденныйЭлемент = Неопределено Тогда
			НовыйРеквизит = Новый РеквизитФормы(ИмяПоля, ТипПоля);
			НовыеРеквизиты.Добавить(НовыйРеквизит); 
		КонецЕсли;
	КонецЦикла;

	ИзменитьРеквизиты(НовыеРеквизиты);
   
    Если НовыеРеквизиты.Количество() = 0 Тогда
    	Возврат;
    КонецЕсли;  
    
	Для НомерПоля=2 По Количество Цикл
		ИмяПоля = "ПолеТовара" + Формат(НомерПоля, "ЧГ=0"); 
		ЭлГруппыТоваров = Элементы.ГруппаПолейТовара; 
		ПолеТовара = Элементы.Добавить(ИмяПоля, ТипЗнч(ШаблонПоля),ЭлГруппыТоваров);
		ЗаполнитьЗначенияСвойств(ПолеТовара, ШаблонПоля, , "Имя, ПутьКДанным, ВыделенныйТекст");
		ПолеТовара.ПутьКДанным = ИмяПоля;
	КонецЦикла;
